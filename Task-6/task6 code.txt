void loop()
{
  unsigned long currentMillis = millis();

  handleConnections();

  if (WiFi.status() == WL_CONNECTED)
  {
    if (Blynk.connected())
    {
      Blynk.run();
    }
  }

  checkSerialCommand();

  if (currentMillis - previousMillis >= monitorInterval)
  {
    previousMillis = currentMillis;

    readCells();
    calculateBattery();
    detectSensorFaults();
    detectBatteryFaults();
    detectFrozenADC();
    detectRelayMismatch();
    manageSystemMode();
    controlOutputs();
    calculateBatteryRisk();
    updateOperatorRecommendation();
    detectTelemetryEvents();
    updateLCD();
    serialOutput();
  }

  processTelemetryQueue();

  if (currentMillis - screenTimer >= screenInterval)
  {
    screenTimer = currentMillis;

    currentScreen++;

    if (currentScreen > 3)
    {
      currentScreen = 0;
    }
  }
}

void handleConnections()
{
  unsigned long currentMillis = millis();

  bool wifiConnected = (WiFi.status() == WL_CONNECTED);

  if (!wifiConnected)
  {
    if (currentMillis - wifiReconnectTimer >= wifiReconnectInterval)
    {
      wifiReconnectTimer = currentMillis;

      Serial.println();
      Serial.println("WiFi disconnected");
      Serial.println("Trying WiFi reconnect...");

      WiFi.disconnect();
      WiFi.begin(ssid, password);
    }

    lastWiFiConnected = false;
    lastBlynkConnected = false;

    return;
  }

  if (!lastWiFiConnected)
  {
    Serial.println();
    Serial.println("WiFi connection restored");

    addTelemetryEvent(EVENT_RECONNECT);

    oldPackVoltage = -999.0;
  }

  if (!Blynk.connected())
  {
    if (currentMillis - blynkReconnectTimer >= blynkReconnectInterval)
    {
      blynkReconnectTimer = currentMillis;

      Serial.println();
      Serial.println("Blynk disconnected");
      Serial.println("Trying Blynk reconnect...");

      Blynk.connect(1000);
    }

    lastBlynkConnected = false;
    lastWiFiConnected = true;

    return;
  }

  if (!lastBlynkConnected)
  {
    Serial.println();
    Serial.println("Blynk connection restored");

    addTelemetryEvent(EVENT_RECONNECT);

    oldSystemMode = (SystemMode)(-1);
    oldFault = (FaultType)(-1);
    oldPackVoltage = -999.0;
  }

  lastWiFiConnected = true;
  lastBlynkConnected = true;
}

void addTelemetryEvent(TelemetryEvent event)
{
  int nextTail = (eventTail + 1) % EVENT_QUEUE_SIZE;

  if (nextTail == eventHead)
  {
    Serial.println("Telemetry queue full");
    return;
  }

  eventQueue[eventTail] = event;
  eventTail = nextTail;

  Serial.print("Telemetry event queued: ");

  if (event == EVENT_STATE_CHANGE)
  {
    Serial.println("STATE CHANGE");
  }
  else if (event == EVENT_FAULT)
  {
    Serial.println("FAULT");
  }
  else if (event == EVENT_VOLTAGE)
  {
    Serial.println("VOLTAGE");
  }
  else if (event == EVENT_RECONNECT)
  {
    Serial.println("RECONNECT");
  }
}

bool getTelemetryEvent(TelemetryEvent *event)
{
  if (eventHead == eventTail)
  {
    return false;
  }

  *event = eventQueue[eventHead];

  eventHead++;

  if (eventHead >= EVENT_QUEUE_SIZE)
  {
    eventHead = 0;
  }

  return true;
}

void detectTelemetryEvents()
{
  bool stateChanged = false;
  bool faultChanged = false;
  bool voltageChanged = false;

  if (systemMode != oldSystemMode)
  {
    stateChanged = true;
  }

  if (activeFault != oldFault)
  {
    faultChanged = true;
  }

  if (fabs(packVoltage - oldPackVoltage) >= VOLTAGE_CHANGE_THRESHOLD)
  {
    voltageChanged = true;
  }

  if (stateChanged)
  {
    addTelemetryEvent(EVENT_STATE_CHANGE);
    oldSystemMode = systemMode;
  }

  if (faultChanged)
  {
    addTelemetryEvent(EVENT_FAULT);
    oldFault = activeFault;
  }

  if (voltageChanged)
  {
    addTelemetryEvent(EVENT_VOLTAGE);
    oldPackVoltage = packVoltage;
  }
}

void readCells()
{
  cell1Voltage = (analogRead(CELL1) / 4095.0) * 4.2;
  cell2Voltage = (analogRead(CELL2) / 4095.0) * 4.2;
  cell3Voltage = (analogRead(CELL3) / 4095.0) * 4.2;
  cell4Voltage = (analogRead(CELL4) / 4095.0) * 4.2;
}

void calculateBattery()
{
  packVoltage =
    cell1Voltage +
    cell2Voltage +
    cell3Voltage +
    cell4Voltage;

  averageVoltage = packVoltage / 4.0;

  highestVoltage = cell1Voltage;
  strongestCell = "CELL1";

  if (cell2Voltage > highestVoltage)
  {
    highestVoltage = cell2Voltage;
    strongestCell = "CELL2";
  }

  if (cell3Voltage > highestVoltage)
  {
    highestVoltage = cell3Voltage;
    strongestCell = "CELL3";
  }

  if (cell4Voltage > highestVoltage)
  {
    highestVoltage = cell4Voltage;
    strongestCell = "CELL4";
  }

  lowestVoltage = cell1Voltage;
  weakestCell = "CELL1";

  if (cell2Voltage < lowestVoltage)
  {
    lowestVoltage = cell2Voltage;
    weakestCell = "CELL2";
  }

  if (cell3Voltage < lowestVoltage)
  {
    lowestVoltage = cell3Voltage;
    weakestCell = "CELL3";
  }

  if (cell4Voltage < lowestVoltage)
  {
    lowestVoltage = cell4Voltage;
    weakestCell = "CELL4";
  }

  if (averageVoltage > 0)
  {
    imbalance =
      ((highestVoltage - lowestVoltage) /
       averageVoltage) * 100.0;
  }
  else
  {
    imbalance = 0;
  }

  if (imbalance < 5)
  {
    batteryStatus = "HEALTHY";
  }
  else if (imbalance < 15)
  {
    batteryStatus = "MINOR IMBALANCE";
  }
  else if (imbalance < 30)
  {
    batteryStatus = "CRITICAL";
  }
  else
  {
    batteryStatus = "PACK FAILURE";
  }
}

void calculateBatteryRisk()
{
  if (systemMode == SHUTDOWN)
  {
    batteryRisk = 3;
    return;
  }

  if (systemMode == FAILSAFE)
  {
    batteryRisk = 3;
    return;
  }

  if (imbalance >= 30)
  {
    batteryRisk = 3;
  }
  else if (imbalance >= 15)
  {
    batteryRisk = 2;
  }
  else if (imbalance >= 5)
  {
    batteryRisk = 1;
  }
  else
  {
    batteryRisk = 0;
  }

  if (systemMode == DEGRADED &&
      batteryRisk < 2)
  {
    batteryRisk = 2;
  }
}

  }
}

void detectSensorFaults()
{
  activeFault = NO_FAULT;

  if (cell1Voltage < 0.1)
  {
    activeFault = CELL1_DISCONNECTED;
    return;
  }

  if (cell1Voltage > 4.3)
  {
    activeFault = INVALID_READING;
    return;
  }

  if (cell2Voltage < 0.1)
  {
    activeFault = CELL2_DISCONNECTED;
    return;
  }

  if (cell2Voltage > 4.3)
  {
    activeFault = INVALID_READING;
    return;
  }

  if (cell3Voltage < 0.1)
  {
    activeFault = CELL3_DISCONNECTED;
    return;
  }

  if (cell3Voltage > 4.3)
  {
    activeFault = INVALID_READING;
    return;
  }

  if (cell4Voltage < 0.1)
  {
    activeFault = CELL4_DISCONNECTED;
    return;
  }

  if (cell4Voltage > 4.3)
  {
    activeFault = INVALID_READING;
    return;
  }
}

void detectBatteryFaults()
{
  if (activeFault != NO_FAULT)
  {
    return;
  }

  if (lowestVoltage < 3.0)
  {
    activeFault = WEAK_CELL;
    return;
  }

  if (highestVoltage > 4.1)
  {
    activeFault = OVER_VOLTAGE;
    return;
  }
}

void detectFrozenADC()
{
  if (simulateFrozenADC)
  {
    activeFault = ADC_FROZEN;
  }
}

void detectRelayMismatch()
{
#if RELAY_FEEDBACK_ENABLED

  int feedback = digitalRead(RELAY_FEEDBACK);

  if (systemMode == NORMAL ||
      systemMode == DEGRADED)
  {
    if (feedback != HIGH)
    {
      activeFault = RELAY_MISMATCH;
    }
  }

  if (systemMode == FAILSAFE ||
      systemMode == SHUTDOWN)
  {
    if (feedback != LOW)
    {
      activeFault = RELAY_MISMATCH;
    }
  }

#endif
}

void manageSystemMode()
{
  if (activeFault == NO_FAULT)
  {
    if (systemMode != NORMAL)
    {
      if (recoveryStart == 0)
      {
        recoveryStart = millis();
      }

      if (millis() - recoveryStart >= recoveryTime)
      {
        systemMode = NORMAL;

        recoveryStart = 0;
        failsafeStart = 0;

        Serial.println();
        Serial.println("RECOVERY COMPLETE");
        Serial.println("SYSTEM MODE : NORMAL");
      }
    }

    return;
  }

  recoveryStart = 0;

  if (activeFault == CELL1_DISCONNECTED ||
      activeFault == CELL2_DISCONNECTED ||
      activeFault == CELL3_DISCONNECTED ||
      activeFault == CELL4_DISCONNECTED)
  {
    if (systemMode != DEGRADED)
    {
      systemMode = DEGRADED;
      logFault(activeFault);
    }

    return;
  }

  if (activeFault == WEAK_CELL)
  {
    if (systemMode != DEGRADED)
    {
      systemMode = DEGRADED;
      logFault(activeFault);
    }

    return;
  }

  if (activeFault == INVALID_READING ||
      activeFault == OVER_VOLTAGE ||
      activeFault == ADC_FROZEN ||
      activeFault == RELAY_MISMATCH)
  {
    if (systemMode != FAILSAFE &&
        systemMode != SHUTDOWN)
    {
      systemMode = FAILSAFE;
      failsafeStart = millis();

      logFault(activeFault);

      Serial.println();
      Serial.println("SYSTEM MODE : FAILSAFE");
      Serial.println("Shutdown timer started");
    }

    if (systemMode == FAILSAFE)
    {
      if (millis() - failsafeStart >= shutdownTime)
      {
        systemMode = SHUTDOWN;

        Serial.println();
        Serial.println("================================");
        Serial.println("FAILSAFE TIMEOUT");
        Serial.println("SYSTEM MODE : SHUTDOWN");
        Serial.println("OUTPUT DISABLED");
        Serial.println("================================");
      }
    }

    return;
  }
}

void controlOutputs()
{
  if (systemMode == NORMAL)
  {
    digitalWrite(RELAY, HIGH);
    digitalWrite(BUZZER, LOW);
  }
  else if (systemMode == DEGRADED)
  {
    digitalWrite(RELAY, HIGH);
    digitalWrite(BUZZER, HIGH);
  }
  else if (systemMode == FAILSAFE)
  {
    digitalWrite(RELAY, LOW);
    digitalWrite(BUZZER, HIGH);
  }
  else if (systemMode == SHUTDOWN)
  {
    digitalWrite(RELAY, LOW);
    digitalWrite(BUZZER, LOW);
  }
}

void updateLCD()
{
  lcd.clear();

  if (systemMode == FAILSAFE)
  {
    lcd.setCursor(0, 0);
    lcd.print("FAILSAFE");

    lcd.setCursor(0, 1);

    if (activeFault == ADC_FROZEN)
    {
      lcd.print("ADC FROZEN");
    }
    else if (activeFault == OVER_VOLTAGE)
    {
      lcd.print("OVER VOLTAGE");
    }
    else if (activeFault == INVALID_READING)
    {
      lcd.print("INVALID DATA");
    }
    else if (activeFault == RELAY_MISMATCH)
    {
      lcd.print("RELAY ERROR");
    }
    else
    {
      lcd.print("OUTPUT OFF");
    }

    return;
  }

  if (systemMode == SHUTDOWN)
  {
    lcd.setCursor(0, 0);
    lcd.print("SYSTEM");

    lcd.setCursor(0, 1);
    lcd.print("SHUTDOWN");

    return;
  }

  if (systemMode == DEGRADED)
  {
    lcd.setCursor(0, 0);
    lcd.print("DEGRADED");

    lcd.setCursor(0, 1);

    if (activeFault == CELL1_DISCONNECTED)
    {
      lcd.print("CELL1 FAULT");
    }
    else if (activeFault == CELL2_DISCONNECTED)
    {
      lcd.print("CELL2 FAULT");
    }
    else if (activeFault == CELL3_DISCONNECTED)
    {
      lcd.print("CELL3 FAULT");
    }
    else if (activeFault == CELL4_DISCONNECTED)
    {
      lcd.print("CELL4 FAULT");
    }
    else if (activeFault == WEAK_CELL)
    {
      lcd.print("WEAK CELL");
    }
    else
    {
      lcd.print("FAULT");
    }

    return;
  }

  if (currentScreen == 0)
  {
    lcd.setCursor(0, 0);
    lcd.print("PACK:");
    lcd.print(packVoltage, 1);
    lcd.print("V");

    lcd.setCursor(0, 1);
    lcd.print("NORMAL");
  }
  else if (currentScreen == 1)
  {
    lcd.setCursor(0, 0);
    lcd.print("C1:");
    lcd.print(cell1Voltage, 1);
    lcd.print(" C2:");
    lcd.print(cell2Voltage, 1);

    lcd.setCursor(0, 1);
    lcd.print("C3:");
    lcd.print(cell3Voltage, 1);
    lcd.print(" C4:");
    lcd.print(cell4Voltage, 1);
  }
  else if (currentScreen == 2)
  {
    lcd.setCursor(0, 0);
    lcd.print("AVG:");
    lcd.print(averageVoltage, 2);

    lcd.setCursor(0, 1);
    lcd.print("IMB:");
    lcd.print(imbalance, 1);
    lcd.print("%");
  }
  else if (currentScreen == 3)
  {
    lcd.setCursor(0, 0);
    lcd.print("BATTERY:");

    lcd.setCursor(0, 1);
    lcd.print(batteryStatus);
  }
}

void checkSerialCommand()
{
  if (Serial.available() > 0)
  {
    char command = Serial.read();

    if (command == 'F' ||
        command == 'f')
    {
      simulateFrozenADC = true;

      Serial.println();
      Serial.println("ADC FROZEN TEST ENABLED");
      Serial.println();
    }

    if (command == 'R' ||
        command == 'r')
    {
      simulateFrozenADC = false;

      activeFault = NO_FAULT;

      recoveryStart = millis();

      if (systemMode == FAILSAFE ||
          systemMode == SHUTDOWN)
      {
        systemMode = FAILSAFE;
        failsafeStart = 0;
      }

      Serial.println();
      Serial.println("FAULT CLEARED");
      Serial.println("RECOVERY STARTED");
      Serial.println();
    }
  }
}

void logFault(FaultType fault)
{
  if (logIndex < MAX_LOGS)
  {
    faultLogs[logIndex].timestamp = millis();
    faultLogs[logIndex].fault = fault;
    faultLogs[logIndex].mode = systemMode;

    logIndex++;
  }

  Serial.println();
  Serial.println("******** FAULT LOG ********");

  Serial.print("Timestamp : ");
  Serial.print(millis());
  Serial.println(" ms");

  Serial.print("Fault : ");
  printFaultName(fault);

  Serial.print("Mode : ");
  printModeName(systemMode);

  Serial.println("****************************");
}

void printFaultName(FaultType fault)
{
  switch (fault)
  {
    case NO_FAULT:
      Serial.println("NO FAULT");
      break;

    case CELL1_DISCONNECTED:
      Serial.println("CELL1 SENSOR DISCONNECTED");
      break;

    case CELL2_DISCONNECTED:
      Serial.println("CELL2 SENSOR DISCONNECTED");
      break;

    case CELL3_DISCONNECTED:
      Serial.println("CELL3 SENSOR DISCONNECTED");
      break;

    case CELL4_DISCONNECTED:
      Serial.println("CELL4 SENSOR DISCONNECTED");
      break;

    case INVALID_READING:
      Serial.println("INVALID READING");
      break;

    case WEAK_CELL:
      Serial.println("WEAK CELL");
      break;

    case OVER_VOLTAGE:
      Serial.println("OVER VOLTAGE");
      break;

    case ADC_FROZEN:
      Serial.println("ADC FROZEN");
      break;

    case RELAY_MISMATCH:
      Serial.println("RELAY MISMATCH");
      break;
  }
}

void printModeName(SystemMode mode)
{
  switch (mode)
  {
    case NORMAL:
      Serial.println("NORMAL");
      break;

    case DEGRADED:
      Serial.println("DEGRADED");
      break;

    case FAILSAFE:
      Serial.println("FAILSAFE");
      break;

    case SHUTDOWN:
      Serial.println("SHUTDOWN");
      break;
  }
}
